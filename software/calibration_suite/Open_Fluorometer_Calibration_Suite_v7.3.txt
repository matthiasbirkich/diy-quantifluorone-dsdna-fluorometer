function fit(rows,mode,groups){const alpha=Number($('alpha').value)||.05,k=Number($('kfactor').value)||3,nfuture=Math.max(1,Math.round(Number($('nfuture').value)||1));const positives=rows.filter(r=>r.y>0).map(r=>r.y),cmin=positives.length?Math.min(...positives):1,positiveS=groups.filter(g=>g.sx>0).map(g=>g.sx),smin=positiveS.length?Math.min(...positiveS):1,sMap=new Map(groups.map(g=>[g.level+'|'+g.y,g.sx||smin]));const weights=rows.map(r=>mode==='OLS'?1:mode==='1/c'?1/(r.y>0?r.y:cmin):mode==='1/c2'?1/((r.y>0?r.y:cmin)**2):1/((sMap.get(r.level+'|'+r.y)||smin)**2));let S=0,Sx=0,Sy=0,Sxx=0,Sxy=0;rows.forEach((r,i)=>{const w=weights[i],x=r.y,Y=r.x;S+=w;Sx+=w*x;Sy+=w*Y;Sxx+=w*x*x;Sxy+=w*x*Y});const den=S*Sxx-Sx*Sx;if(Math.abs(den)<1e-30)throw Error('Regression is singular.');const b=(S*Sxy-Sx*Sy)/den,a=(Sy-b*Sx)/S;let sse=0,wsse=0,sst=0;const my=mean(rows.map(r=>r.x)),calc=rows.map((r,i)=>{const pred=a+b*r.y,res=r.x-pred;sse+=res*res;wsse+=weights[i]*res*res;sst+=(r.x-my)**2;return{...r,pred,res,weight:weights[i]}}),n=rows.length,df=n-2,residual=Math.sqrt(sse/df),rmse=Math.sqrt(sse/n),r2=1-sse/sst,adjr2=1-(1-r2)*(n-1)/df,aic=n*Math.log(Math.max(sse/n,1e-30))+4,bic=n*Math.log(Math.max(sse/n,1e-30))+2*Math.log(n),mx=mean(rows.map(r=>r.y)),sxxRaw=rows.reduce((z,r)=>z+(r.y-mx)**2,0),t=criticalT(df,alpha,true),din=mode==='OLS'?calcDin32645(residual,b,n,mx,sxxRaw,alpha,k,nfuture):null;return{mode,a,b,n,df,residual,rmse,r2,adjr2,aic,bic,wsse,mx,sxxRaw,t,alpha,k,nfuture,din,rows:calc}}

function solve3(A,b){const M=A.map((r,i)=>[...r,b[i]]);for(let k=0;k<3;k++){let p=k;for(let i=k+1;i<3;i++)if(Math.abs(M[i][k])>Math.abs(M[p][k]))p=i;if(Math.abs(M[p][k])<1e-20)return null;[M[k],M[p]]=[M[p],M[k]];const d=M[k][k];for(let j=k;j<4;j++)M[k][j]/=d;for(let i=0;i<3;i++)if(i!==k){const q=M[i][k];for(let j=k;j<4;j++)M[i][j]-=q*M[k][j]}}return[M[0][3],M[1][3],M[2][3]]}
function mandelTest(rows){const X=rows.map(r=>r.y),Y=rows.map(r=>r.x),n=rows.length;if(n<4)return{ok:false,reason:'At least four measurements required.'};const mx=mean(X),my=mean(Y),qxx=X.reduce((z,x)=>z+(x-mx)**2,0);if(!(qxx>0))return{ok:false,reason:'Distinct reference values required.'};const b1=X.reduce((z,x,i)=>z+(x-mx)*(Y[i]-my),0)/qxx,a1=my-b1*mx,rss1=Y.reduce((z,y,i)=>z+(y-(a1+b1*X[i]))**2,0);let sx=0,sx2=0,sx3=0,sx4=0,sy=0,sxy=0,sx2y=0;for(let i=0;i<n;i++){const x=X[i],y=Y[i];sx+=x;sx2+=x*x;sx3+=x*x*x;sx4+=x*x*x*x;sy+=y;sxy+=x*y;sx2y+=x*x*y}const q=solve3([[n,sx,sx2],[sx,sx2,sx3],[sx2,sx3,sx4]],[sy,sxy,sx2y]);if(!q)return{ok:false,reason:'Quadratic regression failed.'};const[a2,b2,c2]=q,rss2=Y.reduce((z,y,i)=>z+(y-(a2+b2*X[i]+c2*X[i]*X[i]))**2,0),df2=n-3;if(df2<=0||rss2<=0)return{ok:false,reason:'Insufficient residual degrees of freedom.'};const delta=Math.max(0,rss1-rss2),F=delta/(rss2/df2),p=1-fCDF(F,1,df2);return{ok:true,a2,b2,c2,rss1,rss2,F,p,df2,linearSufficient:p>=(Number($('alpha').value)||.05)}}
function grubbsTest(vals){const n=vals.length;if(n<3)return{ok:false};const m=mean(vals),ss=sd(vals);if(!(ss>0))return{ok:true,G:0,Gcrit:Infinity,outlier:false,value:vals[0]};let idx=0,G=-1;vals.forEach((v,i)=>{const g=Math.abs(v-m)/ss;if(g>G){G=g;idx=i}});const alpha=Number($('alpha').value)||.05,t=invTCDF(1-alpha/(2*n),n-2),Gcrit=((n-1)/Math.sqrt(n))*Math.sqrt(t*t/(n-2+t*t));return{ok:true,G,Gcrit,outlier:G>Gcrit,value:vals[idx]}}
function shapiroApprox(vals){const x=[...vals].sort((a,b)=>a-b),n=x.length;if(n<3||n>5000)return{ok:false};const xm=mean(x),den=x.reduce((s,v)=>s+(v-xm)**2,0);if(!(den>0))return{ok:false};let m=[];for(let i=1;i<=n;i++)m.push(normInv((i-.375)/(n+.25)));const mn=Math.sqrt(m.reduce((s,v)=>s+v*v,0));m=m.map(v=>v/mn);let num=0;for(let i=0;i<n;i++)num+=m[i]*x[i];const W=Math.min(.999999,Math.max(.000001,num*num/den));let z,p;if(n===3){p=Math.max(0,Math.min(1,(6/Math.PI)*(Math.asin(Math.sqrt(W))-.9553166181245093)))}else if(n<=11){const gamma=-2.273+.459*n;if(gamma-Math.log(1-W)<=0)p=0;else{const y=-Math.log(gamma-Math.log(1-W)),mu=.5440-.39978*n+.025054*n*n-.0006714*n*n*n,sigma=Math.exp(1.3822-.77857*n+.062767*n*n-.0020322*n*n*n);z=(y-mu)/sigma;p=1-normalCDF(z)}}else{const l=Math.log(n),mu=-1.5861-.31082*l-.083751*l*l+.0038915*l*l*l,sigma=Math.exp(-.4803-.082676*l+.0030302*l*l),y=Math.log(1-W);z=(y-mu)/sigma;p=1-normalCDF(z)}return{ok:true,W,p:Math.max(0,Math.min(1,p))}}
function pointSlopes(groups){const g=[...groups].sort((a,b)=>a.y-b.y),rows=[];if(g.length<2)return{ok:false};for(let i=0;i<g.length-1;i++){const dx=g[i+1].y-g[i].y,dy=g[i+1].mx-g[i].mx,local=dx!==0?dy/dx:NaN;rows.push({from:g[i],to:g[i+1],local})}const med=median(rows.map(r=>r.local)),threshold=Math.max(0,Number($('pts-threshold').value)||0);rows.forEach(r=>{r.deviation=med!==0?100*(r.local-med)/med:NaN;r.warning=Number.isFinite(r.deviation)&&Math.abs(r.deviation)>threshold});return{ok:true,median:med,threshold,rows,warnings:rows.filter(r=>r.warning).length}}
