\usepackage{needspace}

% Keep headings together with a reasonable amount of following text
\AddToHook{cmd/section/before}{\Needspace{10\baselineskip}}
\AddToHook{cmd/subsection/before}{\Needspace{8\baselineskip}}
\AddToHook{cmd/subsubsection/before}{\Needspace{6\baselineskip}}

% Avoid single paragraph lines at page boundaries
\clubpenalty=10000
\widowpenalty=10000
\displaywidowpenalty=10000
